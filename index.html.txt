<script>


/************************************************
 * QUAN TRỌNG
 *
 * DÁN LINK WEB APP CHECK DATE VÀO ĐÂY
 *
 * Ví dụ:
 *
 * https://script.google.com/macros/s/ABC123/exec
 *
 ************************************************/

const APP_URL =
  'https://script.google.com/macros/s/AKfycbxlOGbs_cydvONk_Jn4id1KG6CbvQuZhtP3HOvUTz_-8XDhdp9nocI1hsc4YmVUXAub/exec';


let scanner =
  null;


let scanning =
  false;


let alreadyScanned =
  false;


/************************************************
 * KHỞI ĐỘNG
 ************************************************/

window.addEventListener(
  'load',
  function() {

    if (
      APP_URL.indexOf(
        'DÁN_LINK'
      ) >= 0
    ) {

      setStatus(
        'Bạn chưa nhập APP_URL trong file camera.'
      );

      return;

    }


    document
      .getElementById(
        'backLink'
      )
      .href =
        APP_URL;


    startCamera();

  }
);


/************************************************
 * CAMERA
 ************************************************/

function startCamera() {

  scanner =
    new Html5Qrcode(
      'reader'
    );


  const config = {

    fps: 10,

    qrbox:
      function(
        width,
        height
      ) {

        const boxWidth =
          Math.min(
            width * 0.88,
            350
          );


        const boxHeight =
          Math.min(
            height * 0.35,
            160
          );


        return {

          width:
            boxWidth,

          height:
            boxHeight

        };

      },

    formatsToSupport: [

      Html5QrcodeSupportedFormats.EAN_13,

      Html5QrcodeSupportedFormats.EAN_8,

      Html5QrcodeSupportedFormats.UPC_A,

      Html5QrcodeSupportedFormats.UPC_E,

      Html5QrcodeSupportedFormats.CODE_128,

      Html5QrcodeSupportedFormats.CODE_39,

      Html5QrcodeSupportedFormats.ITF

    ]

  };


  /*
   * Dùng camera sau.
   */

  scanner
    .start(

      {
        facingMode:
          {
            ideal:
              'environment'
          }
      },

      config,

      function(
        decodedText
      ) {

        handleBarcode(
          decodedText
        );

      },

      function() {

        /*
         * Không hiển thị lỗi
         * khi từng frame chưa đọc được.
         */

      }

    )

    .then(
      function() {

        scanning =
          true;


        setStatus(
          'Đưa mã vạch vào khung camera.'
        );

      }
    )

    .catch(
      function(error) {

        console.error(
          error
        );


        setStatus(
          'Không thể mở camera. ' +
          'Hãy cho phép Camera cho trang này.'
        );

      }
    );

}


/************************************************
 * XỬ LÝ MÃ VẠCH
 ************************************************/

function handleBarcode(
  code
) {

  if (
    alreadyScanned
  ) {

    return;

  }


  if (!code) {

    return;

  }


  code =
    String(
      code
    ).trim();


  if (!code) {

    return;

  }


  alreadyScanned =
    true;


  setStatus(
    'Đã đọc mã: ' +
    code +
    ' ...'
  );


  if (
    navigator.vibrate
  ) {

    navigator.vibrate(
      150
    );

  }


  /*
   * Dừng camera
   */

  if (
    scanner &&
    scanning
  ) {

    scanner
      .stop()
      .then(
        function() {

          returnToApp(
            code
          );

        }
      )
      .catch(
        function() {

          returnToApp(
            code
          );

        }
      );

  } else {

    returnToApp(
      code
    );

  }

}


/************************************************
 * QUAY VỀ APP
 ************************************************/

function returnToApp(
  barcode
) {

  try {

    const url =
      new URL(
        APP_URL
      );


    url.searchParams.set(
      'barcode',
      barcode
    );


    window.location.href =
      url.toString();

  }

  catch(error) {

    setStatus(
      'Link APP_URL không hợp lệ.'
    );

    alreadyScanned =
      false;

  }

}


/************************************************
 * NHẬP THỦ CÔNG
 ************************************************/

function manualScan() {

  const code =
    document
      .getElementById(
        'manualCode'
      )
      .value
      .trim();


  if (!code) {

    alert(
      'Vui lòng nhập mã vạch.'
    );

    return;

  }


  returnToApp(
    code
  );

}


/************************************************
 * QUAY LẠI
 ************************************************/

function goBack() {

  window.location.href =
    APP_URL;

}


/************************************************
 * THÔNG BÁO
 ************************************************/

function setStatus(
  text
) {

  document
    .getElementById(
      'status'
    )
    .textContent =
      text;

}
